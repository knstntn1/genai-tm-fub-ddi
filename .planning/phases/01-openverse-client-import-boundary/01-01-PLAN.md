```typescript
export const OPENVERSE_IMAGES_URL = 'https://api.openverse.org/v1/images/';
export const DEFAULT_OPENVERSE_PAGE_SIZE = 20;

export interface OpenVerseImageResult {
    id: string;
    title: string;
    imageUrl: string;
    thumbnailUrl: string;
    width?: number;
    height?: number;
    source?: string;
    foreignLandingUrl?: string;
    license?: string;
    licenseUrl?: string;
    creator?: string;
    mature?: boolean;
}

export interface OpenVerseImageSearchResult {
    results: OpenVerseImageResult[];
    page: number;
    pageCount: number;
    pageSize: number;
    resultCount: number;
}

export interface SearchOpenVerseImagesOptions {
    query: string;
    page?: number;
    pageSize?: number;
    signal?: AbortSignal;
}

export type OpenVerseSearchErrorCode =
    | 'empty-query'
    | 'network'
    | 'rate-limited'
    | 'http'
    | 'invalid-response';

export class OpenVerseSearchError extends Error {
    readonly code: OpenVerseSearchErrorCode;
    readonly status?: number;
    readonly retryAfter?: string;
}

export async function searchOpenVerseImages(
    options: SearchOpenVerseImagesOptions
): Promise<OpenVerseImageSearchResult>;
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser to OpenVerse API | Student-entered query leaves the SPA and an untrusted third-party API response enters local code. |
| OpenVerse API response to future UI/importer | Untrusted response fields become app-owned result objects consumed by later phases. |
| future selected result to canvas importer | Third-party image URLs are passed onward but are not loaded by this plan. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-1-01 | I | `src/util/openverse.ts` | mitigate | Do not include `client_id`, `client_secret`, `Authorization`, bearer tokens, or OpenVerse credential env vars; use anonymous `fetch(url.toString(), { signal })`. |
| T-1-02 | D | `searchOpenVerseImages` | mitigate | Use `DEFAULT_OPENVERSE_PAGE_SIZE = 20`, clamp `pageSize` to `1..50`, send `mature=false`, and map `429` plus `Retry-After` to `rate-limited`. |
| T-1-03 | T | OpenVerse response normalization | mitigate | Require string `id`, `url`, and `thumbnail`; drop malformed entries only when valid entries remain; throw `invalid-response` when no valid entry can be trusted. |
| T-1-04 | D | Network and invalid-response handling | mitigate | Convert failed fetches, non-OK responses, and invalid response shapes into `OpenVerseSearchError` codes consumed by later UI; return valid empty result arrays as successful empty results. |
| T-1-05 | T | Third-party image tainting | transfer | This plan does not load images; Plan 02 must set `crossOrigin = 'anonymous'` and validate canvas readback before returning samples. |
| T-1-06 | D | Oversized images | transfer | This plan preserves dimensions only as metadata; Plan 02 must bound canvas dimensions to `maxSize` before returning samples. |
| T-1-07 | T | State mutation before validation | mitigate | This client returns search result data only and imports nothing from `@genaitm/state`; class sample mutation is out of scope for Phase 1. |
</threat_model>
