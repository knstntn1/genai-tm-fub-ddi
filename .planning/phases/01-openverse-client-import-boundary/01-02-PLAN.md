```typescript
export const DEFAULT_OPENVERSE_IMPORT_TIMEOUT_MS = 10000;
export const DEFAULT_OPENVERSE_IMPORT_MAX_SIZE = 512;

export interface ImportOpenVerseImageOptions {
    imageUrl: string;
    fallbackUrl?: string;
    timeoutMs?: number;
    maxSize?: number;
    signal?: AbortSignal;
}

export type OpenVerseImageImportErrorCode =
    | 'load-failed'
    | 'timeout'
    | 'decode-failed'
    | 'canvas-unreadable'
    | 'unsupported-image'
    | 'aborted';

export class OpenVerseImageImportError extends Error {
    readonly code: OpenVerseImageImportErrorCode;
    readonly sourceUrl?: string;
}

export async function importOpenVerseImage(options: ImportOpenVerseImageOptions): Promise<HTMLCanvasElement>;
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| third-party image URL to browser image loader | Untrusted remote media is loaded by the browser and may fail, decode incorrectly, be oversized, or taint canvas. |
| image element to canvas | Remote pixels cross into a canvas that existing sample display/training code expects to read. |
| importer result to future class state mutation | Later phases will insert the returned canvas only after this utility resolves successfully. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-1-08 | I | Browser-exposed OpenVerse credentials | mitigate | This importer accepts URLs only and must not import OpenVerse credentials, Vite secret env vars, `Authorization`, or bearer tokens. |
| T-1-09 | T | Third-party image tainting | mitigate | Set `image.crossOrigin = 'anonymous'` before `src`, draw into a local canvas, and call `context.getImageData(0, 0, 1, 1)` before returning. |
| T-1-10 | D | Oversized images | mitigate | Default `maxSize` to `512` and ensure `Math.max(canvas.width, canvas.height) <= maxSize` before returning. |
| T-1-11 | D | Network/rate-limit/provider failure | mitigate | Convert load failure, timeout, decode failure, abort, and unsupported URL cases into `OpenVerseImageImportError` codes. API rate-limit handling is owned by Plan 01. |
| T-1-12 | T | State mutation before validation | mitigate | Do not import `@genaitm/state`, do not call `setData`, and do not create sample objects; return a canvas only after readability validation passes. |
| T-1-13 | D | Duplicate pending import clicks | mitigate | Keep the utility stateless and deterministic; later UI/integration phases must track pending result IDs and call this function only once per pending result. |
</threat_model>
